\begin{algorithm}
	\scriptsize
	\caption{\label{alg:adaptation_mechanism} Adaptation mechanism}
	\DontPrintSemicolon
	\BlockOfConstants{
		\AdaptTimer \tcp*[1]{waiting time}
	}
	\BlockOfVar{
		\Algo \tcp*[1]{current broadcat protocol}
		\AlgoId \tcp*[1]{protocol identifier}
		$\MapT[]$ \tcp*[1]{map of timers}
		$\MapT[\MsgPidExt].\RemTime$ \tcp*[1]{remaining time of timer}
		$\MapT[\MsgPidExt].\MsgContent$ \tcp*[1]{content of message}
	}
	

	
	\Fn(\tcp*[f]{on reception of message $\mathit{\Ms}$}){\FuncHandler{$m$: \Msg}}{
		\eIf{\AlgoId $=$ \MsgPid}{
			\If{$\MapT[\MsgPid]  \neq \bot$}{
				\textbf{cancel}($\MapT[\MsgPid]$) \;
			}
			\Algo.$\AlgoBroadcast($m)
		}{
		\If{$\MapT[\MsgPid] = \bot$}{
			$\MapT[\MsgPid] \gets \TimerFunc(\AdaptTimer, \Ms)$
		}
	}
}

\Fn(\tcp*[f]{case: $\mathit{\MapT[\MsgPidExt].\RemTime = 0}$}){\FuncExp{$t$: \Timer}}{
	$\mathit{m' \gets \AlgoInit(t.\MsgContent)}$\;
	$\mathit{\AlgoEmit(m')}$
}
\end{algorithm}
