\begin{algorithm}
	\scriptsize
	\caption{\label{alg:adaptation_mechanism} Adaptation mechanism}
	\DontPrintSemicolon
	\BlockOfVar{
		$\mathit{\AdaptTimer, \Algo}$ \tcp*[1]{waiting time; running protocol}
		$\mathit{\MsgPidExt}$ \tcp*[1]{message of protocol with identifier pid}
		$\mathit{\AlgoId, \MapT}$ \tcp*[1]{protocol identifier; map of timers}
		$\mathit{\MapT[\MsgPidExt].\RemTime}$ \tcp*[1]{remaining time of timer}
		$\mathit{\MapT[\MsgPidExt].\MsgContent}$ \tcp*[1]{content of message}
	}
	
	\BlockOfFuncs{
		$\mathit{\AlgoInit(c: \MsgContent)}$ \tcp*[1]{create a new message}
		$\mathit{\AlgoEmit(m: \Msg)}$ \tcp*[1]{send message as source node}
		$\mathit{\AlgoBroadcast(m: \Msg)}$ \tcp*[1]{implementation of running protocol}
	}
	
	\Fn{\FuncHandler{$m$: \Msg}}{
		\eIf{$\mathit{\AlgoId = \MsgPid}$}{
			\If{$\mathit{\MapT[\MsgPid] \neq \bot}$}{
				$\mathit{\textbf{cancel}(\MapT[\MsgPid])}$ \;
				$\mathit{\MapT[\MsgPid]} \gets \bot$
			}
			$\mathit{\AlgoBroadcast(m)}$
		}{
		\If{$\mathit{\MapT[\MsgPid] = \bot}$}{
			$\mathit{\MapT[\MsgPid] \gets \TimerFunc(\AdaptTimer, \Ms)}$
		}
	}
}
\tcp*[1]{case: $\mathit{\MapT[\MsgPidExt].\RemTime = 0}$}
\Fn{\FuncExp{$t$: \Timer}}{
	$\mathit{m' \gets \AlgoInit(t.\MsgContent)}$\;
	$\mathit{\AlgoEmit(m')}$
}
\end{algorithm}

\begin{algorithm}
	\scriptsize
	\caption{\label{alg:flooding} Two flooding approaches}
	\DontPrintSemicolon
	\BlockOfVar{
		$\mathit{\FloodTimer}$ \tcp*[1]{waiting time}		
		$\mathit{\FloodMode}$ \tcp*[1]{bimodal way of flooding (\FloodSimp or \FloodCoun)}
		$\mathit{\FloodRecpLi}$ \tcp*[1]{maximum number of receptions per message}
		$\mathit{\SentMsgs, \MapFlo}$ \tcp*[1]{set of dispatched messages; map of timers}
		$\mathit{\MapFlo[\MsgIdExt].\RecNo}$ \tcp*[1]{number of times a message is received}
		$\mathit{\MapFlo[\MsgIdExt].\RemTime}$ \tcp*[1]{remaining time of timer}
	}
	
	\BlockOfAlgoAPI{
		$\mathit{\AlgoSend(m: \Msg)}$ \tcp*[1]{send a message}
	}
	
	\Fn{\AlgoBroFunc{m: \Msg}}{
		\Switch{\FloodMode}{
			\Case{\FloodSimp}{
				\If{$\mathit{\Ms.\Id \notin \SentMsgs}$}{
					$\mathit{\AlgoSend(\Ms)}$ \;
					$\mathit{\SentMsgs \gets \SentMsgs \cup \{\Ms.\Id\}}$ \;
				}
			}
			\Case{\FloodCoun}{
				\If{$\mathit{\Ms.\Id \notin \SentMsgs}$}{
					\eIf{$\mathit{\MapFlo[\Ms.\Id] = \bot}$}{
						$\mathit{\MapFlo[\Ms.\Id] \gets \TimerFunc(\FloodTimer, \Ms)}$ \;
						$\mathit{\MapFlo[\Ms.\Id].\RecNo \gets 1}$ \;
					}{
					$\mathit{\MapFlo[\Ms.\Id].\RecNo \gets \MapFlo[\Ms.\Id].\RecNo + 1}$ \;
					\If{$\mathit{\MapFlo[\Ms.\Id].\RecNo = \FloodRecpLi}$}{
						$\mathit{\textbf{cancel}(\MapFlo[\Ms.\Id])}$ \;
						$\mathit{\SentMsgs \gets \SentMsgs \cup \{\Ms.\Id\}}$ \;
					}
					
				}
			}		
		}
	}
}
\tcp*[1]{case $\mathit{\MapFlo[\MsgIdExt].\RemTime} = 0$}
\Fn{\FuncExpFlo{m: \Msg}}{
	$\mathit{\AlgoSend(\Ms)}$ \;
	$\mathit{\SentMsgs \gets \SentMsgs \cup \{\Ms.\Id\}}$ \;
}

\Fn{\AlgoInitFunc{c: \MsgContent}}{
	$\mathit{\Ms \gets \FloodNewMsgFun()}$ \;
	$\mathit{\Ms.\MsgContent \gets c}$ \;
	\textbf{return} $\mathit{\Ms}$
}

\Fn{\AlgoEmitFunc{m: \Msg }}{
	$\mathit{\AlgoSend(\Ms)}$ \;
	$\mathit{\SentMsgs \gets \SentMsgs \cup \{\Ms.\Id\}}$ \;
}

\end{algorithm}
