\begin{algorithm}
	\footnotesize
	\DontPrintSemicolon
	\raziel{implementation detail: in text mention that the set "delivered" must be cleaned periodically to avoid collecting an $\infty$ number of delivered messages.} \;
	\BlockOfVar{
		\FloodMode \tcp*[1]{type of dissemination to perform}
		\Delivered \tcp*[1]{log of messages delivered to application}
		\BlockOfFloodCounter{
			\MapC[] \tcp*[1]{map of counters}
			\BlockOfMapAttributes{
				$\MapC[\MsgId].\Receptions$ \tcp*[1]{receptions of message \MsgId}
			}
			\BlockOfConstants{
				\FloodRecpLi \tcp*[1]{number of allowed receptions}
			}
		}
		\BlockOfFloodTimeBased{
			\BlockOfConstants{
				\AdaptTimer \tcp*[1]{default waiting time}
			}
		}
	}
	
	\Fn{\AlgoHandle{\Ms: \Msg}}{
		\If{ $\Ms.\Id \notin \Delivered$ }{
			\Switch{\FloodMode}{
				\Case{\FloodSimp}{
					$\AlgoSend(\Ms)$ \;
					$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$ \;
				}
				\Case{\FloodCount}{
					\eIf{$\MapC[\MsgId].\Receptions < \FloodRecpLi$}{
						$\AlgoSend(\Ms)$ \;
						$\MapC[\MsgId].\Receptions \gets \MapC[\MsgId].\Receptions + 1$
					}{
						$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$ \;
					}
				}
				\Case{\FloodFixed}{
					$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$ \;
					$\MapT \gets \TimerFunc(\AdaptTimer)$ \tcp*[1]{create new timer}
					\BlockOfTimerExpires{
						$\AlgoSend(\Ms)$ \;
					}
				}
				\Case{\FloodRandom}{
					$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$ \;
					\tcp*[1]{get a random number between 0 and \AdaptTimer}
					$\R \gets \RandomFunc([0,\AdaptTimer])$ \;
					$\MapT \gets \TimerFunc(\R)$ \tcp*[1]{create new timer}
					\BlockOfTimerExpires{
						$\AlgoSend(\Ms)$ \;
					}
				}
			}
		}
	}
	
	\Fn{\AlgoInit{\MsgPayload: \MsgContent}}{
		$\Ms' \gets \FloodNewMsg$ \;
		$\Ms'.\MsgContent \gets$ pl \;
		\textbf{return} $\Ms'$ \;
	}
	
	\Fn{\AlgoEmit{\Ms: \Msg }}{
		$\Delivered \gets \Delivered \cup \{\MsgId\}$ \;
		$\AlgoSend(\Ms)$
	}
	\caption{\labalg{typesOfFlooding}Group of basic flooding protocols}
\end{algorithm}

		}
	}

\end{algorithm}

\begin{algorithm}
	\footnotesize
	\DontPrintSemicolon
	
	\BlockOfConstants{
		\AdaptTimer \tcp*[1]{default waiting time} \label{alg:adaptation_mechanism:timeout}
	}

	\BlockOfVar{
		\Algo		\tcp*[1]{local instance of broadcast protocol}
		\AlgoPid  	\tcp*[1]{protocol identifier}
		$\MapT[]$ 	\tcp*[1]{map from message ids to timers}
		
		\BlockOfMapAttributes{
			$\MapT[ \MsgId ].\RemTime$	\tcp*[1]{remaining time}
			$\MapT[ \MsgId ].\MsgContent$ 	\tcp*[1]{content of message}
		}
	}

	\Fn(\tcp*[f]{on message reception}){\FuncHandler{\Ms: \Msg}}{
		\eIf{ \AlgoPid $=$ \MsgPid }{ \label{alg:adaptation_mechanism:eval_ids}
			\If{ $\MapT[\MsgId]  \neq \bot$ }{
				$\MapT[\MsgId] \gets \bot$ 	\tcp*[1]{cancel timer}
			}
			\Algo.$\AlgoHandle(\Ms)$ \label{alg:adaptation_mechanism:handle}
		}{
			\If{ $\MapT[\MsgId] = \bot$ }{
				$\MapT[\MsgId] \gets \TimerFunc$ \tcp*[1]{create new timer}
				$\MapT[\MsgId].\RemTime \gets \AdaptTimer$ \label{alg:adaptation_mechanism:set_timer} \;
				$\MapT[\MsgId].\MsgContent \gets \Ms.\MsgContent$ \tcp*[1]{keep message payload}
				\BlockOfMapTimerExpires{
					$\Ms' \gets \Algo.\AlgoInit(\MapT[\MsgId].\MsgContent)$ \;
					$\Algo.\AlgoEmit(\Ms')$ \label{alg:adaptation_mechanism:emit}
				}
			}
		}
	}

\begin{algorithm}
	\footnotesize
	\DontPrintSemicolon
	
	\BlockOfVar{
		\NodeId \;
		\Relay \;
		\Delivered \;
		\Neigs \;
		\ObsNeigs \;
		\ObsRelays \;
	}
	
	\Fn{\AlgoHandle{\Ms: \Msg}}{
		\If{ $\Ms.\Id \notin \Delivered$ }{
			\Switch{\MsgType}{
				\Case{\MsgCtrl}{
					$\ObsNeigs \gets \ObsNeigs \cup \{\MsgEmitter\}$ \;
					\If{$\MsgEmitRel = \T$}{
						$\ObsRelays \gets \ObsRelays \cup \{\MsgEmitter\}$
					}
				}
				\Case{\MsgBro}{
					\If{$\Relay = \T$}{
						$\AlgoSend(\Ms)$ \;
						$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$
					}
				}
			}
		}
	}
	
	\Fn(\tcp*[f]{periodic deamon to send control messages}){\Demon{}}{
		$\CtrlMsg \gets \CtrlMsgFun$ \;
		$\CtrlMsg.\Emitter \gets \NodeId$ \;
		$\CtrlMsg.\IsRelay\gets \Relay$ \;
		$\AlgoSend(\CtrlMsg)$
	}
	
	\Fn(\tcp*[f]{periodic update of relay status}){\Update{}}{
		\eIf{$\Neigs \neq \ObsNeigs$}{
			$\Neigs \gets \ObsNeigs$ \;
			$\Relay \gets \Rule(\Neigs)$ \;
			$\ObsNeigs \gets \emptyset$ \;
			$\ObsRelays \gets \emptyset$ \;	
		}{
			$\Relay \gets \Min(\ObsRelays)$ \;
		}
	}
\caption{\labalg{overlay-based}Functioning of overlay-based algorithms}
\end{algorithm}

\begin{algorithm}
	\footnotesize
	\DontPrintSemicolon
	
	\BlockOfVar{
		\Delivered \;
		$\MapT[]$ 	\tcp*[1]{map from message ids to timers}
		
		\BlockOfMapAttributes{
			$\MapT[ \MsgId ].\RemTime$	\tcp*[1]{remaining time}
			$\MapT[ \MsgId ].\MsgContent$ 	\tcp*[1]{content of message}
		}
	}
	
	\Fn(\tcp*[f]{on message reception}){\AlgoHandle{\Ms: \Msg}}{
		\If{ $\Ms.\Id \notin \Delivered$ }{
			$\Fwd \gets \LocRule(\MsgPos)$ \;
			\eIf{$\Fwd = \T$}{
				\If{ $\MapT[\MsgId] \neq \bot$ }{
					$\MapT[\MsgId] \gets \bot$ \tcp*[1]{cancel timer}
				}
				$\MsgPos \gets \GetPo()$ \;
				$\AlgoSend(\Ms)$ \;
				$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$
			}{
				$\MapT[\MsgId] \gets \TimerFunc$ \tcp*[1]{create new timer}
				$\MapT[\MsgId].\RemTime \gets \Timeout(\MsgPos)$ \;
				$\MapT[\MsgId].\MsgContent \gets \Ms.\MsgContent$ \tcp*[1]{keep message payload}
				\BlockOfMapTimerExpires{
					$\MsgPos \gets \GetPo()$ \;
					$\AlgoSend(\Ms)$ \;
					$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$
				}
			}
		}
	}

\caption{\labalg{location-based}Functioning of location-based algorithms}
\end{algorithm}
