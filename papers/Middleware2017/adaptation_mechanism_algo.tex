%\Comment{fields in t are: msgId and timer}

\begin{algorithm}
\caption{Adaptation mechanism}\label{adapMech}
\begin{algorithmic}[1]
\Function{onMsgReception}{$M$}
	\State $id \gets$ \textbf{concat}$(M.id, M.algoId)$
	\If{$M.algoId = local.algoId$}
		\State $cancelTimerIfAny(M.id)$
		\State $local.handleMsg(M.id, M.payload)$
		\State \textbf{return}
	\Else
		\State /*First reception of a foreign message*/
		\If{$id \notin rcvMsgs$}
			\State $rcvMsgs.add(id)$
			\State $t \gets (M.id,$ \textbf{newTimer}$(local.T))$
			\State $rcvPayloads.add(t)$
			\State \textbf{call} $doEvent(M)$ \textbf{when} $t$ \textbf{expires}
		\EndIf
	\EndIf \label{endIf}
\EndFunction

\Function{doEvent}{$M$}
	\State $M.algoId \gets local.algoId$
	\State $local.handleMsg(M.id, M.payload)$
\EndFunction

\Function{cancelTimerIfAny}{$messageId$}
	\State $t \gets (messageId,$ \textbf{newTimer}$(local.T))$
	\If{$t \in rcvPayloads$}
	\State $timer \gets rcvPayloads.get(t)$
	\State \textbf{cancelEvent}$(timer)$
	\EndIf
\EndFunction
\end{algorithmic}
\end{algorithm}
