\begin{algorithm}
	\footnotesize
	\DontPrintSemicolon
	
	\BlockOfVar{
		\FloodMode \tcp*[1]{type of dissemination to perform}
		\Delivered \tcp*[1]{messages already delivered to application}
		\BlockOfFloodCounter{
			\MapC[] \tcp*[1]{map of counters}
			\BlockOfMapAttributes{
				$\MapC[\MsgId].\Receptions$ \tcp*[1]{receptions of message \MsgId}
			}
			\BlockOfConstants{
				\FloodRecpLi \tcp*[1]{number of allowed receptions}
			}
		}
		\BlockOfFloodTimeBased{
			\BlockOfConstants{
				\AdaptTimer \tcp*[1]{default waiting time}
			}
		}
	}
	
	\Fn{\AlgoHandle{\Ms: \Msg}}{
		\If{ $\Ms.\Id \notin \Delivered$ }{
			\Switch{\FloodMode}{
				\Case{\FloodSimp}{
					$\AlgoSend(\Ms)$ \;
					$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$ \;
				}
				\Case{\FloodCount}{
					\eIf{$\MapC[\MsgId].\Receptions < \FloodRecpLi$}{
						$\AlgoSend(\Ms)$ \;
						$\MapC[\MsgId].\Receptions \gets \MapC[\MsgId].\Receptions + 1$
					}{
						$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$ \;
					}
				}
				\Case{\FloodFixed}{
					$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$ \;
					$\MapT \gets \TimerFunc(\AdaptTimer)$ \tcp*[1]{create new timer}
					\BlockOfTimerExpires{
						$\AlgoSend(\Ms)$ \;
					}
				}
				\Case{\FloodRandom}{
					$\Delivered \gets \Delivered \cup \{\Ms.\Id\}$ \;
					\tcp*[1]{get a random number between 0 and \AdaptTimer}
					$\R \gets \RandomFunc([0,\AdaptTimer])$ \;
					$\MapT \gets \TimerFunc(\R)$ \tcp*[1]{create new timer}
					\BlockOfTimerExpires{
						$\AlgoSend(\Ms)$ \;
					}
				}
			}
		}
	}
	
	\Fn{\AlgoInit{\MsgPayload: \MsgContent}}{
		$\Ms' \gets \FloodNewMsg$ \;
		$\Ms'.\MsgContent \gets$ pl \;
		\textbf{return} $\Ms'$ \;
	}
	
	\Fn{\AlgoEmit{\Ms: \Msg }}{
		$\Delivered \gets \Delivered \cup \{\MsgId\}$ \;
		\er{can insert wait here}
		\raziel{I think that we can take rid off emit(), it looks redundant now that the presentation of flooding algorithms is within one algorithm. Otherwise, I do have to repeat the SWITCH shown in handle()}
		$\AlgoSend(\Ms)$ \;
	}
	\caption{\labalg{typesOfFlooding}Group of basic flooding protocols}
\end{algorithm}

\begin{algorithm}
	\footnotesize
	\DontPrintSemicolon
	
	\BlockOfVar{
		\Algo		\tcp*[1]{local instance of broadcast protocol}
		\AlgoPid  	\tcp*[1]{protocol identifier}
	}

	\Fn(\tcp*[f]{on message reception}){\FuncHandler{\Ms: \Msg}}{
		\eIf{ \AlgoPid $=$ \MsgPid }{ \label{alg:adaptation_mechanism_simple:eval_ids}
			\Algo.$\AlgoHandle(\Ms)$ \label{alg:adaptation_mechanism_simple:handle} \;
		}{
			$\Ms' \gets \Algo.\AlgoInit(\MapT[\MsgId].\MsgContent)$ \;
			$\Algo.\AlgoEmit(\Ms')$ \label{alg:adaptation_mechanism_simple:emit} \;
		}
	}
\caption{\labalg{adaptation_mechanism_simple}Simple adaptation mechanism}
\end{algorithm}

\begin{algorithm}
	\footnotesize
	\DontPrintSemicolon
	
	\BlockOfConstants{
		\AdaptTimer \tcp*[1]{default waiting time} \label{alg:adaptation_mechanism:timeout}
	}

	\BlockOfVar{
		\Algo		\tcp*[1]{local instance of broadcast protocol}
		\AlgoPid  	\tcp*[1]{protocol identifier}
		$\MapT[]$ 	\tcp*[1]{map from message ids to timers}
		
		\BlockOfMapAttributes{
			$\MapT[ \MsgId ].\RemTime$	\tcp*[1]{remaining time}
			$\MapT[ \MsgId ].\MsgContent$ 	\tcp*[1]{content of message}
		}
	}

	\Fn(\tcp*[f]{on message reception}){\FuncHandler{\Ms: \Msg}}{
		\eIf{ \AlgoPid $=$ \MsgPid }{ \label{alg:adaptation_mechanism:eval_ids}
			\If{ $\MapT[\MsgId]  \neq \bot$ }{
				$\MapT[\MsgId] \gets \bot$ 	\tcp*[1]{cancel timer}
			}
			\Algo.$\AlgoHandle(\Ms)$ \label{alg:adaptation_mechanism:handle}
		}{
			\If{ $\MapT[\MsgId] = \bot$ }{
				$\MapT[\MsgId] \gets \TimerFunc$ \tcp*[1]{create new timer}
				$\MapT[\MsgId].\RemTime \gets \AdaptTimer$ \label{alg:adaptation_mechanism:set_timer} \;
				$\MapT[\MsgId].\MsgContent \gets \Ms.\MsgContent$ \tcp*[1]{keep message payload}
			}
		}
	}

	\Fn(\tcp*[f]{triggers when $\MapT[\MsgId].\RemTime = 0$}){\FuncExp{$\MapT[\MsgId]$: \Timer}}{
		$\Ms' \gets \Algo.\AlgoInit(\MapT[\MsgId].\MsgContent)$ \;
		$\Algo.\AlgoEmit(\Ms')$ \label{alg:adaptation_mechanism:emit}
	}
\caption{\labalg{adaptation_mechanism}Efficient adaptation mechanism}
\end{algorithm}
