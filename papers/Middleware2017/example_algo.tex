\begin{algorithm}
	\scriptsize
	\caption{\label{alg:adaptation_mechanism}Adaptation mechanism}
	\DontPrintSemicolon
	\BlockOfVar{
		$\mathit{\Algo, \MapT, default\_time}$ \tcp*[1]{running protocol; map of timers; waiting time}
		$\mathit{\Algo.id}$ \tcp*[1]{protocol identifier}
		$\mathit{\Algo.init(payload)}$ \tcp*[1]{create a new message}
		$\mathit{\Algo.send(\Msg)}$ \tcp*[1]{send message as source node}
		$\mathit{\Algo.broadcast(\Msg)}$ \tcp*[1]{broadcast message using running protocol}
		$\mathit{\MapT[\MsgPid].time}$ \tcp*[1]{remaining time of timer}
		$\mathit{\MapT[\MsgPid].content}$ \tcp*[1]{content of message from protocol pid}
	}
	\Fn{\FuncHandler{$m$: \Msg}}{
		\eIf{$\mathit{\Algo.id = \MsgPid}$}{
			\If{$\mathit{\MapT[\MsgPid] \neq \bot}$}{
				$\mathit{\textbf{cancel}(\MapT[\MsgPid])}$ \;
				$\mathit{\MapT[\MsgPid]} \gets \bot$
			}
			$\mathit{\Algo.broadcast(m)}$
		}{
			\If{$\mathit{\MapT[\MsgPid] = \bot}$}{
				$\mathit{\MapT[\MsgPid] \gets newTimer(default\_time, m)}$
			}
		}
	}
	
	\Fn{\FuncExp{$timer$: \MapT[\MsgPid]}}{
		$\mathit{m' \gets \Algo.init(timer.content)}$\;
		$\mathit{\Algo.send(m')}$
	}
\end{algorithm}
