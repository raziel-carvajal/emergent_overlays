\begin{algorithm}
\footnotesize
\DontPrintSemicolon
\caption{Interoperability mechanism to deal with more than one running protocol in a MANET.}
\label{interop-alg}
\vars{
	% VARIABLES
	node\_id \textit{: string} \tcp*[1]{node identifier}
	running\_alg \textit{: string} \tcp*[1]{running algorithm}
	timeout\textit {: double} \tcp*[1]{maximum waiting time to start the procedure of border dectection}
	am\_i\_border \textit{: boolean} \tcp*[1]{says wether the local node is a border node or not}

	% SETS
	known\_algs \textit{: set} \tcp*[1]{previously known foreign protocols}

	% MAPS
	timers \textit{: dictionary} \tcp*[1]{timers of border detection procedure}
	known\_nodes \textit{: dictionary} \tcp*[1]{previously known nodes from a foreign protocol}
}

\func{ \receive{msg} \label{interop-alg:rcv-fun} }{ 
	\eIf{ msg.sender\_alg = running\_alg \Ork am\_i\_border }{
		broadcast(msg) \; \label{interop-alg:bro}
	}{
		\Switch{msg}{
			\Case{ ControlMsg }{

				known\_nodes[msg.sender\_alg] $\gets$ msg.sender\_id \; \label{interop-alg:save-sender}
				
				\If{ msg.sender\_alg $\notin$ known\_algs }{
				
					known\_algs $\gets$ known\_algs $\cup$ msg.sender\_alg \;	\label{interop-alg:save-alg}
					delay $\gets$ $new$ Random(0, timeout) \;					
					timers[msg.sender\_alg] $\gets$ $new$ Timer(delay) \; \label{interop-alg:new-timer}
					
					\TimerExpires{

						b $\gets$ $new$ BorderMsg() \; \label{interop-alg:new-border-msg}
						b.chosen\_node $\gets$ known\_nodes[msg.sender\_alg].pickOne() \label{interop-alg:new-border-msg:chosen} \;
						b.fwd\_no $\gets$ MAX\_FWD\_NO \; \label{interop-alg:new-border-msg:hops}
						b.sender\_id $\gets$ node\_id \; \label{interop-alg:new-border-msg:sender}						
						b.sender\_alg $\gets$ running\_alg \; \label{interop-alg:new-border-msg:sender-alg}
						b.foster\_by $\gets$ msg.sender\_alg \; \label{interop-alg:new-border-msg:alg}
						forward(b) \; \label{interop-alg:fwd}

					}
				}
			}
			\Case{ BorderMsg }{
			
				msg.fwd\_no $\gets$ msg.fwd\_no - 1 \; \label{interop-alg:border-msg-case:dec}
				
				\If{ msg.fwd\_no $\neq$ 0 }{
					forward(msg) \; \label{interop-alg:border-msg-case:fwd}
				}
				
				\If{ running\_alg = msg.foster\_by \And msg.foster\_by $\notin$ known\_algs }{
				
					known\_algs $\gets$ known\_algs $\cup$ msg.foster\_by \; \label{interop-alg:border-msg-case:update-known-algs-1}

					\If{ node\_id = msg.chosen\_node }{					
						am\_i\_border $\gets$ True \; \label{interop-alg:border-msg-case:select-as-border}
					}
					
				}
				
				\If{ running\_alg = msg.sender\_alg \And msg.sender\_alg $\notin$ known\_algs }{
				
					known\_algs $\gets$ known\_algs $\cup$ msg.sender\_alg \; \label{interop-alg:border-msg-case:update-known-algs-2}

					\If{ timers[msg.sender\_alg] $\neq$ $\emptyset$ }{
						cancelTimer(timers[msg.sender\_alg]) \; \label{interop-alg:border-msg-case:rm-timers}
					}
					
				}
					
			}
		}
	}
}
\end{algorithm}
