\begin{algorithm}[t]
\begin{footnotesize}
	\begin{algorithmic}[1]
	
	\BeginConstants
		\State $\Delta$: time \Comment{period of sending of \texttt{hello} messages for MPR}
	\EndConstants
	
	\BeginVariables
		\State b: boolean, init \texttt{false} \Comment{is the node a border node?}
		\State t: time, init $\bot$ \Comment{only meaningful if b is \texttt{true}}
	\EndVariables
	
	\smallskip
	
	\Procedure{amIBorder}{ }
		\If{b $\wedge$ (currentTime - t) $> 2\times\Delta$}
			b $\gets$ \texttt{false}
		\EndIf
		\State return b
	\EndProcedure
	
	\smallskip
	
	\begin{center}
	{\textbf{Code for node currently using controlled flooding}}
	\end{center}
	\vspace{-2.5mm}
	\hrulefill
		
	\Procedure{receive}{MPR \texttt{hello} from $n_j$}
		\State $t_{n_j} \gets $ new timer with duration $\in [10:100]$ms
	\EndProcedure
	
	\Procedure{receive}{\texttt{border}$(n_j)$ from $n_k$ }
		\State cancel $t_{n_j}$ (if it exists)
	\EndProcedure
	
	\Procedure{expires}{$t_{n_j}$}
		\State b $\gets$ \texttt{true} ; t $\gets$ currentTime \Comment{now a border node for $2\times \Delta$s}
		\State MAC.localBroadcast(\texttt{border}($n_j$))
	\EndProcedure
	
	\Procedure{receive}{message $m$ for the first time}
		\State CF.receive($m$) \Comment{Regular handling of new incoming message}
		\If{amIBorder()}
			\State $t_m \gets \infty$ \Comment{Border node always send $m$ when timer expires}
		\EndIf
	\EndProcedure
	
	\smallskip
	
	\begin{center}
	{\textbf{Code for node currently using MPR}}
	\end{center}
	\vspace{-2.5mm}
	\hrulefill
	
	\Procedure{receive}{\texttt{border}$(n_j)$ from $n_k$}
		\State b $\gets$ \texttt{true} ; t $\gets$ currentTime \Comment{now a border node for $2\times \Delta$s}
	\EndProcedure
	
	\Procedure{receive}{message $m$ for the first time, from $n_k$}
		\If{$n_k$ uses controlled flooding}
			\State TODO, should I act immediately as a source, or wait? Should I do it only if I am border? Should I declare myself as a border?
		\Else \Comment{$n_k$ uses MPR}
			\If{$m$.relays includes $n_i$}
				\State b $\gets$ \texttt{false}
			\Else \Comment{$n_j$ is not an overlay relay}
				\If{amIBorder()}
					CF.receive($m$) \Comment{use controlled flooding}
				\EndIf
			\EndIf
			\State MPR.receive($m$) \Comment{Regular handling of new incoming message}
		\EndIf
	\EndProcedure

\end{algorithmic}	
\end{footnotesize}
\caption{Interoperability mechanisms on node $n_i$}\label{algorithm:interoperability:cf}
\end{algorithm}
