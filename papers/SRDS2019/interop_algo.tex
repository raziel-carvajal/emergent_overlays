\begin{algorithm}[t]
\begin{footnotesize}
	\begin{algorithmic}[1]
	
	\BeginConstants
		\State $\Delta$: time \Comment{period of sending of \texttt{hello} messages for MPR}
	\EndConstants
	
	\BeginVariables
		\State b: boolean, init \texttt{false} \Comment{should the node act as a border node?}
		\State t: time, init $\bot$ \Comment{only meaningful if b is \texttt{true}}
	\EndVariables
	
	\Procedure{amIBorder}{ }
		\If{b $\wedge$ (currentTime - t) $> 2\times\Delta$}
			b $\gets$ \texttt{false}
		\EndIf
		\State return b
	\EndProcedure
	
	\smallskip
	
	\begin{center}
	{\textbf{Code for node currently using controlled flooding} (CF)}
	\end{center}
	\vspace{-2.5mm}
	\hrulefill
	
	\Procedure{MAC.receive}{MPR \texttt{hello} from $n_j$}
		\State $t_{n_j} \gets $ new timer with duration $\in [10:100]$ms
	\EndProcedure
	
	\Procedure{CF.receive}{\texttt{border}$(n_j)$ from $n_k$ }
		\State cancel $t_{n_j}$ (if it exists)
	\EndProcedure
	
	\BeginTimerExpires{$t_{n_j}$}
		\State b $\gets$ \texttt{true} ; t $\gets$ currentTime \Comment{now a border node for $2\times \Delta$ s}
		\State MAC.localBroadcast(\texttt{border}($n_j$))
	\EndTimerExpires
	
	\Procedure{CF.receive}{message $m$ for the first time}
		\State CF.handle($m$) \Comment{regular handling of the message}
		\If{amIBorder()}
			$\text{counter}_m \gets \infty$ \Comment{forces being a relay}
		\EndIf
	\EndProcedure
	
	\smallskip
	
	\begin{center}
	{\textbf{Code for node currently using MPR}}
	\end{center}
	\vspace{-2.5mm}
	\hrulefill
	
	\Procedure{MAC.receive}{\texttt{border}$(n_j)$ from $n_k$}
		\State b $\gets$ \texttt{true} ; t $\gets$ currentTime \Comment{now a border node for $2\times \Delta$s}
	\EndProcedure
	
	\Procedure{MPR.receive}{message $m$ for the first time}
		\If{$m$ was sent using MPR}
			\If{$m$.relays includes $n_i$}
				b $\gets$ \texttt{false}
			\Else \Comment{$n_i$ is not part of the overlay}
				\If{amIBorder()}
					CF.handle($m$) \Comment{if border, use CF}
				\EndIf
			\EndIf
			\State MPR.handle($m$) \Comment{regular handling of the message}
		\Else \Comment{$m$ received from a CF node}
			\State MPR.broadcast($m$) \Comment{initiate MPR broadcast}
		\EndIf
	\EndProcedure

\end{algorithmic}
\end{footnotesize}
\caption{Interoperability mechanisms on node $n_i$}\label{algorithm:interoperability}
\end{algorithm}
