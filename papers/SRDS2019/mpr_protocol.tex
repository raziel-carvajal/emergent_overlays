\begin{algorithm}[t]
	\begin{algorithmic}[1]
	
		\BeginVariables
			\State me : \emph{string} \Comment{node identifier}
			\State oneHopNeigs : \emph{set}
			\State twoHopNeigs[ ] : \emph{dictionary}
			\BeginMapDetails
				\State \emph{key} : \emph{string}
				\State \emph{value} : \emph{set}
			\EndMapDetails
			
		\EndVariables
		
		\Procedure{choseRelays}{}
			\State relays $\gets \emptyset$ 
			\State uncovered $\gets \emptyset$
			\For{set \textbf{in} twoHopNeigs.\Call{values}{ } }
				\State uncovered $\gets$ uncovered $\cup$ set
			\EndFor
			\For{n \textbf{in} oneHopNeigs}
				\State $s = \{u, me\}$
				\If{$\exists u \in$ uncovered \textbf{such} twoHopNeigs[n] = $s$ }
					\State relays $\gets$ relays $\cup \{u\}$
					\State \textbf{delete} $u$ \textbf{from} uncovered
				\EndIf
			\EndFor
			\Repeat
				\BeginSelectN \textbf{in} oneHopNeigs \textbf{when}
				  \State $\nexists n' \in$  oneHopNeigs \textbf{such} $s_{1} > s_{2}$ 
 					\BeginWith
 						\State $s_{1} \gets |$ twoHopNeigs[n] $|$
 						\State $s_{2} \gets |$ twoHopNeigs[$n'$] $|$
 					\EndWith
 					\State relays $\gets$ relays $\cup \{n\}$
 					\State \textbf{delete} twoHopNeigs[n] \textbf{from} uncovered
 					\State \textbf{delete} n \textbf{from} oneHopNeigs
				\EndSelectN
			\Until{uncovered $\neq \emptyset$ \textbf{or} oneHopNeigs $\neq \emptyset$}
			\State \Return relays
		\EndProcedure
	\end{algorithmic}	
\caption{Selection of relays with MPR technique}\label{algorithm:mpr}
\end{algorithm}
